139struct MotorContext {
142 tmcl::MotorType motor_type = tmcl::MotorType::BLDC_MOTOR;
143
146 uint8_t pole_pairs = 1;
147
149 tmcl::VelocitySensorSelection velocity_sensor =
150 tmcl::VelocitySensorSelection::SAME_AS_COMMUTATION;
151
154 uint32_t encoder_cpr = 0;
155
160 [[nodiscard]] constexpr uint32_t cpr() const noexcept {
161 const uint32_t pp = (pole_pairs == 0u) ? 1u : pole_pairs;
162 switch (velocity_sensor) {
163 case tmcl::VelocitySensorSelection::SAME_AS_COMMUTATION:
164 return 65536u * pp;
165 case tmcl::VelocitySensorSelection::DIGITAL_HALL:
166 return 6u * pp;
167 case tmcl::VelocitySensorSelection::ABN1_ENCODER:
168 case tmcl::VelocitySensorSelection::ABN2_ENCODER:
169 case tmcl::VelocitySensorSelection::SPI_ENCODER:
170 return encoder_cpr;
171 }
172 return 0u;
173 }
174
182 [[nodiscard]] double k_rpm() const noexcept {
183 return static_cast<double>(cpr()) * detail::kVelScalePerCpr;
184 }
185
187 [[nodiscard]] bool valid() const noexcept { return cpr() != 0u; }
188};

195[[nodiscard]] inline double velocityToRpm(double value, VelocityUnit unit,
196 MotorContext const& ctx) noexcept {
197 switch (unit) {
198 case VelocityUnit::Rpm: return value;
199 case VelocityUnit::Rps: return value * detail::kRpmPerRps;
200 case VelocityUnit::RadPerSec: return value * detail::kRpmPerRadPerSec;
201 case VelocityUnit::DegPerSec: return value * detail::kRpmPerDegPerSec;
202 case VelocityUnit::Internal: {
203 const double k = ctx.k_rpm();
204 return (k > 0.0) ? (value / k) : 0.0;
205 }
206 }
207 return 0.0;
208}

211[[nodiscard]] inline double velocityFromRpm(double rpm, VelocityUnit unit,
212 MotorContext const& ctx) noexcept {
213 switch (unit) {
214 case VelocityUnit::Rpm: return rpm;
215 case VelocityUnit::Rps: return rpm / detail::kRpmPerRps;
216 case VelocityUnit::RadPerSec: return rpm / detail::kRpmPerRadPerSec;
217 case VelocityUnit::DegPerSec: return rpm / detail::kRpmPerDegPerSec;
218 case VelocityUnit::Internal: return rpm * ctx.k_rpm();

233[[nodiscard]] inline int32_t velocityToInternal(double value, VelocityUnit unit,
234 MotorContext const& ctx) noexcept {
235 const double internal_d = velocityFromRpm(velocityToRpm(value, unit, ctx),
236 VelocityUnit::Internal, ctx);
237 if (internal_d > static_cast<double>(detail::kVelocityInternalMax)) return detail::kVelocityInternalMax;
238 if (internal_d < static_cast<double>(detail::kVelocityInternalMin)) return detail::kVelocityInternalMin;
239 return static_cast<int32_t>(internal_d >= 0.0 ? internal_d + 0.5 : internal_d - 0.5);
240}

242// ============================================================================
243// Acceleration conversions
244//
245// `AccelerationUnit::Internal` is the **raw RAMPER_A register value** as
246// stored in NR 54..59 (RAMP_AMAX/A1/A2/DMAX/D1/D2). The chip's scaling is:
247// RAMPER_A_register = (dV_internal/dt) * 2^17 / fCLK
248// (Parameter Mode reference manual, p.72; Register Mode p.44).
249// ============================================================================
250
251[[nodiscard]] inline double accelerationToRpmPerSec(double value, AccelerationUnit unit,
252 MotorContext const& ctx) noexcept {
253 switch (unit) {
254 case AccelerationUnit::RpmPerSec: return value;
255 case AccelerationUnit::RpsPerSec: return value * detail::kRpmPerRps;
256 case AccelerationUnit::RadPerSec2: return value * detail::kRpmPerRadPerSec;
257 case AccelerationUnit::DegPerSec2: return value * detail::kRpmPerDegPerSec;
258 case AccelerationUnit::Internal: {
259 // value is the register integer; recover dV/dt in internal
260 // velocity units per second, then divide by k_rpm to get RPM/s.
261 const double k = ctx.k_rpm();
262 if (k <= 0.0 || detail::kAccelInternalPerVdt <= 0.0) return 0.0;
263 const double dvdt_internal = value / detail::kAccelInternalPerVdt;
264 return dvdt_internal / k;
265 }
266 }
267 return 0.0;
268}
269
270[[nodiscard]] inline double accelerationFromRpmPerSec(double rpm_per_sec, AccelerationUnit unit,
271 MotorContext const& ctx) noexcept {
272 switch (unit) {
273 case AccelerationUnit::RpmPerSec: return rpm_per_sec;
274 case AccelerationUnit::RpsPerSec: return rpm_per_sec / detail::kRpmPerRps;
275 case AccelerationUnit::RadPerSec2: return rpm_per_sec / detail::kRpmPerRadPerSec;
276 case AccelerationUnit::DegPerSec2: return rpm_per_sec / detail::kRpmPerDegPerSec;
277 case AccelerationUnit::Internal: {
278 // Forward direction: RAMPER_A_register = rpm/s * k_rpm * (2^17 / fCLK)
279 return rpm_per_sec * ctx.k_rpm() * detail::kAccelInternalPerVdt;
280 }
281 }
282 return 0.0;
283}

292[[nodiscard]] inline uint32_t accelerationToInternal(double value, AccelerationUnit unit,
293 MotorContext const& ctx) noexcept {
294 // Convert to the raw RAMPER_A register integer and saturate to the
295 // chip's unsigned 23-bit range (1 .. 2^23-1).
296 const double reg_d = accelerationFromRpmPerSec(
297 accelerationToRpmPerSec(value, unit, ctx), AccelerationUnit::Internal, ctx);
298 if (reg_d <= 0.0) return detail::kAccelInternalMin;
299 if (reg_d > static_cast<double>(detail::kAccelInternalMax)) return detail::kAccelInternalMax;
300 const uint32_t reg = static_cast<uint32_t>(reg_d + 0.5);
301 return (reg < detail::kAccelInternalMin) ? detail::kAccelInternalMin : reg;
302}

352[[nodiscard]] inline double angleToDegElec(double value, AngleUnit unit,
353 MotorContext const& ctx) noexcept {
354 const double pp = (ctx.pole_pairs == 0u) ? 1.0 : static_cast<double>(ctx.pole_pairs);
355 switch (unit) {
356 case AngleUnit::DegElec: return value;
357 case AngleUnit::RadElec: return value * (180.0 / detail::kPi);
358 case AngleUnit::DegMech: return value * pp;
359 case AngleUnit::RadMech: return value * pp * (180.0 / detail::kPi);
360 case AngleUnit::PhiERaw: return value * (360.0 / detail::kPhiEPerElecRev);
361 }
362 return 0.0;
363}
364
365[[nodiscard]] inline double angleFromDegElec(double deg_elec, AngleUnit unit,
366 MotorContext const& ctx) noexcept {
367 const double pp = (ctx.pole_pairs == 0u) ? 1.0 : static_cast<double>(ctx.pole_pairs);
368 switch (unit) {
369 case AngleUnit::DegElec: return deg_elec;
370 case AngleUnit::RadElec: return deg_elec * (detail::kPi / 180.0);
371 case AngleUnit::DegMech: return deg_elec / pp;
372 case AngleUnit::RadMech: return deg_elec / pp * (detail::kPi / 180.0);
373 case AngleUnit::PhiERaw: return deg_elec * (detail::kPhiEPerElecRev / 360.0);
374 }
375 return 0.0;
376}

378[[nodiscard]] inline double convertAngle(double value, AngleUnit from, AngleUnit to,
379 MotorContext const& ctx) noexcept {
380 if (from == to) return value;
381 return angleFromDegElec(angleToDegElec(value, from, ctx), to, ctx);
382}

384[[nodiscard]] inline int16_t angleToPhiE(double value, AngleUnit unit,
385 MotorContext const& ctx) noexcept {
386 double phi_d = angleFromDegElec(angleToDegElec(value, unit, ctx), AngleUnit::PhiERaw, ctx);
387 // Wrap into [-32768, 32767]
388 while (phi_d > 32767.0) phi_d -= 65536.0;
389 while (phi_d < -32768.0) phi_d += 65536.0;
390 return static_cast<int16_t>(phi_d >= 0.0 ? phi_d + 0.5 : phi_d - 0.5);
391}
