372 static bool fromSpi(std::span<const uint8_t, 8> in, TMCLReply& r, uint8_t sent_opcode = 0,
373 uint16_t sent_type = 0) noexcept {
374 // Store raw bytes for advanced parsing (ALWAYS, even if parsing fails)
375 std::copy(in.begin(), in.end(), r.rawBytes.begin());
376
377 // Check for SESSION_START status codes (bootloader or parameter mode)
378 uint8_t raw_status = in[0];
379 if (raw_status == 0x13) {
380 // SESSION_START from bootloader (0x13)
381 // These use different protocols, so TMCL checksum will fail
382 // But we still want to return success so caller can use rawBytes
383 r.spi_status = static_cast<SPIStatus>(raw_status);
384 r.status = in[1];
385 r.opcode = in[2];
386 r.value = (static_cast<uint32_t>(in[3]) << 24) | (static_cast<uint32_t>(in[4]) << 16) |
387 (static_cast<uint32_t>(in[5]) << 8) | static_cast<uint32_t>(in[6]);
388 return true; // Allow SESSION_START even if checksum fails
389 }
390
391 // Check for GetVersion string format (Command 136, Type 0)
392 // GetVersion with Type=0 returns a special string format without checksum
393 // Format: [Host Address][Version String 8 chars]
394 if (sent_opcode == 136 && sent_type == 0) {
395 // We sent GetVersion Type=0, expect string format reply (no checksum validation)
396 r.spi_status = SPIStatus::OK; // Assume OK for version string
397 r.status = 100; // REPLY_OK for successful version retrieval
398 r.opcode = 136; // GetVersion command
399 r.value = 0; // No value field for string format
400 return true; // Success - skip checksum validation for string format
401 }
402
403 // Standard TMCL reply validation
404 if (tmclChecksum(in.data(), 7) != in[7])
405 return false;
406 r.spi_status = static_cast<SPIStatus>(in[0]);
407 if (r.spi_status == SPIStatus::NOT_READY || r.spi_status == SPIStatus::CHECKSUM_ERROR)
408 return false;
409 r.status = in[1];
410 r.opcode = in[2];
411 r.value = (static_cast<uint32_t>(in[3]) << 24) | (static_cast<uint32_t>(in[4]) << 16) |
412 (static_cast<uint32_t>(in[5]) << 8) | static_cast<uint32_t>(in[6]);
413 return true;
414 }

433 static bool fromUart(std::span<const uint8_t, 9> in, uint8_t addr, TMCLReply& r,
434 uint8_t sent_opcode = 0, uint16_t sent_type = 0) noexcept {
435 // Store raw bytes (map 9-byte UART to 8-byte format, skip first byte)
436 std::copy(in.begin() + 1, in.end(), r.rawBytes.begin());
437
438 // UART Reply Format (per datasheet):
439 // byte0: Host Address (8 bits)
440 // byte1: Sync bit (bit 0) + Module Address (bits 1-7)
441 // byte2: TMCL Status
442 // byte3: Operation
443 // byte4-7: Data (big-endian)
444 // byte8: Checksum
445
446 // Check for GetVersion string format (Command 136, Type 0)
447 // GetVersion with Type=0 returns a special string format without valid checksum
448 // Format: [Host Address][Sync+Address][Version String 7 chars]
449 if (sent_opcode == 136 && sent_type == 0) {
450 // We sent GetVersion Type=0, expect string format reply (no checksum validation)
451 r.spi_status = SPIStatus::OK; // Assume OK for version string
452 r.status = 100; // REPLY_OK for successful version retrieval
453 r.opcode = 136; // GetVersion command
454 r.value = 0; // No value field for string format
455 return true; // Success - skip checksum validation for string format
456 }
457
458 // Verify module address in byte 1 (bits 1-7)
459 // "The module address reuses the upper 7 bits of the bootloader device address"
460 // Compare the upper 7 bits of byte 1 with the upper 7 bits of the address
461 if ((in[1] & 0xFE) != (addr & 0xFE))
462 return false;
463
464 // Verify checksum over first 8 bytes
465 if (tmclChecksum(in.data(), 8) != in[8])
466 return false;
467
468 r.spi_status = SPIStatus::OK; // UART has no SPI status field
469 r.status = in[2]; // TMCL Status at byte 2
470 r.opcode = in[3]; // Operation at byte 3
471 r.value = (static_cast<uint32_t>(in[4]) << 24) | (static_cast<uint32_t>(in[5]) << 16) |
472 (static_cast<uint32_t>(in[6]) << 8) | static_cast<uint32_t>(in[7]);
473 return true;
474 }

499 void toSpi(std::span<uint8_t, 8> out) const noexcept {
500 out[0] = opcode;
501 out[1] = static_cast<uint8_t>(type & 0xFF);
502 out[2] = static_cast<uint8_t>(((type & 0xF00u) >> 4) | (motor & 0x0Fu));
503 out[3] = static_cast<uint8_t>(value >> 24);
504 out[4] = static_cast<uint8_t>(value >> 16);
505 out[5] = static_cast<uint8_t>(value >> 8);
506 out[6] = static_cast<uint8_t>(value);
507 out[7] = tmclChecksum(out.data(), 7);
508 }

515 void toUart(uint8_t addr, std::span<uint8_t, 9> out) const noexcept {
516 // Byte 0: [Sync bit (bit 0)] [Module Address (bits 1-7)]
517 // "The module address reuses the upper 7 bits of the bootloader device address"
518 // This means we take bits 7-1 of the address and place them in bits 7-1 of byte 0
519 // Then set bit 0 (sync bit) to 1
520 out[0] = (addr & 0xFE) | 0x01; // Keep upper 7 bits of address, set sync bit
521 out[1] = opcode;
522 // Bytes 2-3: 12-bit TYPE field + 4-bit MOTOR/BANK field, packed exactly like the
523 // Trinamic reference (`examples/TMC9660/TMC9660.c::sendRequestUART`):
524 // data[2] = type & 0xFF; // type[7:0]
525 // data[3] = (type >> 8) << 4 | (index & 0x0F); // type[11:8] hi-nibble | motor lo-nibble
526 // The previous packing dropped type[11:8] entirely and put motor in the low
527 // nibble, which silently aliased every parameter NR >= 256 down into the
528 // 0..255 range. That made the gate-driver overcurrent / VGS / supply / temp
529 // SAPs (NR 254..299) write the wrong target on the chip, and made the read
530 // scan return totally bogus values for NRs 256..334.
531 out[2] = static_cast<uint8_t>(type & 0xFFu);
532 out[3] = static_cast<uint8_t>(((type & 0xF00u) >> 4) | (motor & 0x0Fu));
533 out[4] = static_cast<uint8_t>(value >> 24);
534 out[5] = static_cast<uint8_t>(value >> 16);
535 out[6] = static_cast<uint8_t>(value >> 8);
536 out[7] = static_cast<uint8_t>(value);
537 out[8] = tmclChecksum(out.data(), 8);
538 }

622[[nodiscard]] constexpr TmclSpiRxClass classifyTmclSpiReply(std::span<const uint8_t, 8> rx,
623 uint8_t expected_opcode,
624 bool verify_opcode) noexcept {
625 if (rx[0] == static_cast<uint8_t>(SPIStatus::NOT_READY)) {
626 return TmclSpiRxClass::NOT_READY;
627 }
628 if (!tmclReplyEchoesOpcode(expected_opcode) || rx[0] == 0x13U) {
629 return TmclSpiRxClass::VALID;
630 }
631 if (rx[0] != static_cast<uint8_t>(SPIStatus::OK)) {
632 return TmclSpiRxClass::STALE;
633 }
634 if (tmclChecksum(rx.data(), 7) != rx[7]) {
635 return TmclSpiRxClass::STALE;
636 }
637 if (verify_opcode && rx[2] != expected_opcode) {
638 return TmclSpiRxClass::STALE;
639 }
640 return TmclSpiRxClass::VALID;
641}

1165 bool transferTMCLBatch(std::span<const TMCLFrame> requests, std::span<TMCLReply> replies,
1166 uint8_t /*address*/, size_t* completed = nullptr) noexcept {
1167 const size_t n = requests.size();
1168 if (completed != nullptr) {
1169 *completed = 0;
1170 }
1171 if (replies.size() < n) {
1172 return false;
1173 }
1174 if (n == 0) {
1175 return true;
1176 }
1177 this->tmclStats_.accesses += static_cast<uint32_t>(n);
1178 const TmclSpiTiming& timing = this->tmclTiming_;
1179 TMCLFrame reply_request{};
1180 reply_request.opcode = kReplyRequestOpcode;
1181
1182 size_t k = 0; // datagram index: request k, or the reply-request when k == n
1183 bool discard = true; // datagram k's reply is not needed (k == 0, or after a recovery)
1184 while (k <= n) {
1185 std::array<uint8_t, 8> tx_buf{};
1186 std::array<uint8_t, 8> rx_buf{};
1187 (k < n ? requests[k] : reply_request).toSpi(tx_buf);
1188 if (!discard && timing.frameGapUs > 0U) {
1189 this->delayUs(timing.frameGapUs);
1190 }
1191 if (!sendUntilAccepted(tx_buf, rx_buf)) {
1192 return false;
1193 }
1194 if (!discard) {
1195 const TMCLFrame& prev = requests[k - 1];
1196 if (classifyTmclSpiReply(rx_buf, prev.opcode, timing.verifyReplyOpcode) ==
1197 TmclSpiRxClass::VALID) {
1198 (void)TMCLReply::fromSpi(rx_buf, replies[k - 1], prev.opcode, prev.type);
1199 } else {
1200 ++this->tmclStats_.staleReplies;
1201 ++this->tmclStats_.recoveries;
1202 if (!exchangeSingle(prev, replies[k - 1], nullptr, nullptr)) {
1203 return false;
1204 }
1205 if (completed != nullptr) {
1206 *completed = k;
1207 }
1208 if (k == n) {
1209 break;
1210 }
1211 discard = true; // the chip now holds the reply-request's reply; re-send request k
1212 continue;
1213 }
1214 if (completed != nullptr) {
1215 *completed = k;
1216 }
1217 }
1218 discard = false;
1219 ++k;
1220 }
1221 return true;
1222 }

1233 const TmclSpiTiming& timing = this->tmclTiming_;
1234 for (uint16_t polls = 0;; ++polls) {

1236 ++this->tmclStats_.failures;

1240 return true;
1241 }

1244 ++this->tmclStats_.failures;

1246 "NOT_READY: re-send budget (%u) exhausted", timing.maxPolls);
1247 return false;

1255 const TMCLFrame* second_command) noexcept {
1256 const TmclSpiTiming& timing = this->tmclTiming_;
1257 std::array<uint8_t, 8> tx_buf{};
1258 std::array<uint8_t, 8> rx1_buf{};
1259 std::array<uint8_t, 8> tx2_buf{};
1260 std::array<uint8_t, 8> rx_buf{};
1261 tx.toSpi(tx_buf);
1262
1263 /* Opcode 0xFF, not 0x00 — see the @warning on transferTMCL(). */
1264 TMCLFrame cmd2{};
1265 cmd2.opcode = kReplyRequestOpcode;
1266 if (second_command != nullptr) {
1267 cmd2 = *second_command;
1268 }
1269 cmd2.toSpi(tx2_buf);
1270
1271 TMC9660_LOG_TMCL_RAW_FRAME(*static_cast<Derived*>(this), 2, "SPI_TMCL",
1272 "[TMCL TX 1 ] %02X %02X %02X %02X %02X %02X %02X %02X", tx_buf[0],
1273 tx_buf[1], tx_buf[2], tx_buf[3], tx_buf[4], tx_buf[5], tx_buf[6],
1274 tx_buf[7]);
1275
1276 // Transactions 1 and 2. Transaction 1 is re-sent while it comes back NOT_READY.
1277 if constexpr (requires(Derived& d, std::array<uint8_t, 8>& a, uint32_t g) {
1278 d.hostSpiTransferTMCLPair(a, a, a, a, g);
1279 }) {
1280 for (uint16_t polls = 0;; ++polls) {
1281 this->tmclStats_.frames += 2U;
1282 if (!static_cast<Derived*>(this)->hostSpiTransferTMCLPair(tx_buf, rx1_buf, tx2_buf, rx_buf,
1283 timing.frameGapUs)) {
1284 ++this->tmclStats_.failures;
1285 return false;
1286 }
1287 if (rx1_buf[0] != static_cast<uint8_t>(SPIStatus::NOT_READY)) {
1288 break;
1289 }
1290 ++this->tmclStats_.notReady;
1291 if (polls >= timing.maxPolls) {
1292 ++this->tmclStats_.failures;
1293 return false;
1294 }
1295 this->delayUs(timing.pollWaitUs(polls));
1296 }
1297 } else {
1298 if (!sendUntilAccepted(tx_buf, rx1_buf)) {
1299 return false;
1300 }
1301 if (timing.frameGapUs > 0U) {
1302 this->delayUs(timing.frameGapUs);
1303 }
1304 if (!sendFrame(tx2_buf, rx_buf)) {
1305 ++this->tmclStats_.failures;
1306 return false;
1307 }
1308 }
1309
1310 TMC9660_LOG_TMCL_RAW_FRAME(*static_cast<Derived*>(this), 2, "SPI_TMCL",
1311 "[TMCL RX 1 ] %02X %02X %02X %02X %02X %02X %02X %02X", rx1_buf[0],
1312 rx1_buf[1], rx1_buf[2], rx1_buf[3], rx1_buf[4], rx1_buf[5],
1313 rx1_buf[6], rx1_buf[7]);
1314
1315 // Optionally capture first reply (this is the reply to the PREVIOUS command due to SPI delay)
1316 if (first_reply != nullptr) {
1317 (void)TMCLReply::fromSpi(rx1_buf, *first_reply);
1318 }
1319
1320 // Poll transaction 2 until it carries a valid reply to tx. Only the reply-request (or the
1321 // caller's second_command) is re-sent, never tx, so a write cannot be duplicated.
1322 for (uint16_t polls = 0;; ++polls) {
1323 TMC9660_LOG_TMCL_RAW_FRAME(*static_cast<Derived*>(this), 2, "SPI_TMCL",
1324 "[TMCL RX 2 ] %02X %02X %02X %02X %02X %02X %02X %02X", rx_buf[0],
1325 rx_buf[1], rx_buf[2], rx_buf[3], rx_buf[4], rx_buf[5], rx_buf[6],
1326 rx_buf[7]);
1327 const TmclSpiRxClass cls = classifyTmclSpiReply(rx_buf, tx.opcode, timing.verifyReplyOpcode);
1328 if (cls == TmclSpiRxClass::VALID) {
1329 break;
1330 }
1331 if (cls == TmclSpiRxClass::NOT_READY) {
1332 ++this->tmclStats_.notReady;
1333 } else {
1334 ++this->tmclStats_.staleReplies;
1335 }
1336 // A well-formed reply that echoes the reply-request opcode means the chip has already
1337 // answered transaction 2 itself, so the reply to tx was consumed by an earlier (stale)
1338 // datagram and will not arrive: fail now instead of polling out the whole budget.
1339 const bool reply_lost =
1340 cls == TmclSpiRxClass::STALE && second_command == nullptr &&
1341 tx.opcode != kReplyRequestOpcode && rx_buf[0] == static_cast<uint8_t>(SPIStatus::OK) &&
1342 rx_buf[2] == kReplyRequestOpcode && tmclChecksum(rx_buf.data(), 7) == rx_buf[7];
1343 if (reply_lost || polls >= timing.maxPolls) {
1344 ++this->tmclStats_.failures;
1345 TMC9660_LOG_DEBUG(*static_cast<Derived*>(this), 1, "SPI_TMCL",
1346 "No valid reply after %u re-sends (last SPI status 0x%02X%s)",
1347 static_cast<unsigned>(polls), static_cast<unsigned>(rx_buf[0]),
1348 reply_lost ? ", reply lost" : "");
1349 std::copy(rx_buf.begin(), rx_buf.end(), reply.rawBytes.begin());
1350 reply.spi_status = static_cast<SPIStatus>(rx_buf[0]);
1351 reply.status = rx_buf[1];
1352 reply.opcode = rx_buf[2];
1353 reply.value = (static_cast<uint32_t>(rx_buf[3]) << 24) |
1354 (static_cast<uint32_t>(rx_buf[4]) << 16) |
1355 (static_cast<uint32_t>(rx_buf[5]) << 8) | static_cast<uint32_t>(rx_buf[6]);
1356 return false;
1357 }
1358 this->delayUs(timing.pollWaitUs(polls));
1359 if (!sendFrame(tx2_buf, rx_buf)) {
1360 ++this->tmclStats_.failures;
1361 return false;
1362 }
1363 }
1364
1365 // Parse reply with command context (for handling special reply formats like GetVersion string)
1366 return TMCLReply::fromSpi(rx_buf, reply, tx.opcode, tx.type);
1367 }

1369protected:
