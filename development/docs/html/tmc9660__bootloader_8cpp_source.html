43inline bool IsExpectedOptionalInfoFailure(uint8_t cmd, uint32_t value, uint8_t status) noexcept {
44 const auto st = static_cast<BootloaderStatus>(status);
45 const bool is_expected_status = (st == BootloaderStatus::INVALID_VALUE) ||
46 (st == BootloaderStatus::INVALID_BANK) ||
47 (st == BootloaderStatus::MEM_UNCONFIGURED) ||
48 (st == BootloaderStatus::CMD_NOT_AVAILABLE);
49 if (!is_expected_status) {
50 return false;
51 }
52
53 if (cmd == static_cast<uint8_t>(BootloaderCommand::GET_INFO)) {
54 return IsOptionalInfoQuery(value);
55 }
56
57 if (cmd == static_cast<uint8_t>(BootloaderCommand::SET_BANK)) {
58 const auto bank = static_cast<MemoryBank>(value);
59 return (bank == MemoryBank::SPI_FLASH) || (bank == MemoryBank::I2C_EEPROM);
60 }
61
62 return false;
63}

108template <typename CommType>
109bool TMC9660Bootloader<CommType>::sendCommand(uint8_t cmd, uint32_t value, uint32_t* reply) noexcept {
110 // Dispatch to appropriate protocol based on interface type
111 // Use compile-time type checking to only instantiate the appropriate function
112 if constexpr (std::is_base_of_v<SpiCommInterface<CommType>, CommType>) {
113 if (comm_.mode() == CommMode::SPI) {
114 return sendCommandSPI(cmd, value, reply);
115 }
116 }
117 if constexpr (std::is_base_of_v<UartCommInterface<CommType>, CommType>) {

120 }
121 }
122 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Unsupported interface mode for bootloader");
123 return false;

140template <typename CommType>
141bool TMC9660Bootloader<CommType>::sendCommandSPI(uint8_t cmd, uint32_t value, uint32_t* reply) noexcept {
142 if (comm_.mode() != CommMode::SPI) {
143 return false;
144 }
145 // At this point, we know comm_.mode() == CommMode::SPI, so CommType must inherit from
146 // SpiCommInterface<CommType>. We cast to the CRTP base class to access the public method,
147 // which will use CRTP (static_cast<Derived*>(this)) internally to call the derived implementation.
148 // This is safe because:
149 // 1. Runtime check ensures mode() == SPI, so CommType is an SPI interface
150 // 2. CommType (e.g., Esp32SPITMC9660CommInterface) inherits from SpiCommInterface<CommType>
151 // 3. The CRTP base class method will properly dispatch to the derived implementation
152 BootloaderCommandSPI tx{cmd, value};
153
154 TMC9660_LOG_DEBUG(comm_, 4, "TMC9660Bootloader",
155 "Sending SPI bootloader command: cmd=0x%02X, value=0x%08X", cmd, value);
156
157 std::array<uint8_t, 5> txBuf, rxBuf;
158 tx.toBuffer(txBuf);
159
160 // Same verbosity as [BL TX NOOP]: raw 5-byte command frame (host→device) for SPI bring-up.
161 TMC9660_LOG_DEBUG(comm_, 4, "TMC9660Bootloader", "[BL TX CMD ] %02X %02X %02X %02X %02X",
162 txBuf[0], txBuf[1], txBuf[2], txBuf[3], txBuf[4]);
163
164 // Call the method directly on comm_ - since CommType inherits from SpiCommInterface<CommType>,
165 // it has spiTransferBootloader() available through inheritance. The CRTP base class method
166 // will internally use static_cast<Derived*>(this) to call the derived implementation.
167 // Step 1: Send the command
168 // NOTE: In SPI, the reply to THIS command will come in the NEXT transaction.
169 // The rxBuf here contains the reply from the PREVIOUS command (we ignore it).
170 if (!comm_.spiTransferBootloader(txBuf, rxBuf)) {
171 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to send SPI command (cmd=0x%02X)",
172 cmd);
173 return false;
174 }
175
176 // TMC9660_LOG_DEBUG(comm_,2, "TMC9660Bootloader", "[BL RX PREV] %02X %02X %02X %02X %02X
177 // (ignored)",
178 // rxBuf[0], rxBuf[1], rxBuf[2], rxBuf[3], rxBuf[4]);
179
180 // Step 2: Send NO_OP command to clock out the reply to our command
181 // This is standard SPI behavior - replies are delayed by one transaction.
182 std::array<uint8_t, 5> dummyTx = {
183 static_cast<uint8_t>(BootloaderCommand::NO_OP), // NO_OP command
184 0, 0, 0, 0 // Dummy value (all zeros)
185 };
186 std::array<uint8_t, 5> replyBuf;
187
188 TMC9660_LOG_DEBUG(comm_, 4, "TMC9660Bootloader", "[BL TX NOOP] %02X %02X %02X %02X %02X",
189 dummyTx[0], dummyTx[1], dummyTx[2], dummyTx[3], dummyTx[4]);
190
191 // Step 2: Send NO_OP to receive the reply to our command
192 if (!comm_.spiTransferBootloader(dummyTx, replyBuf)) {
193 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to receive SPI reply (cmd=0x%02X)",
194 cmd);
195 return false;
196 }
197
198 TMC9660_LOG_DEBUG(comm_, 4, "TMC9660Bootloader", "[BL RX RPLY] %02X %02X %02X %02X %02X",
199 replyBuf[0], replyBuf[1], replyBuf[2], replyBuf[3], replyBuf[4]);
200
201 // Parse reply from the second transaction
202 BootloaderReplySPI rep = BootloaderReplySPI::fromBuffer(replyBuf);
203
204 if (!rep.isOK()) {
205 if (IsExpectedOptionalInfoFailure(cmd, value, rep.status)) {
206 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader",
207 "Optional bootloader info unavailable: status=0x%02X (cmd=0x%02X, query=0x%08X)",
208 rep.status, cmd, value);
209 } else {
210 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader",
211 "SPI command failed: status=0x%02X (cmd=0x%02X)", rep.status, cmd);
212 }
213 return false;
214 }
215
216 if (reply)
217 *reply = rep.value;
218
219 TMC9660_LOG_DEBUG(comm_, 4, "TMC9660Bootloader",
220 "SPI command successful: status=0x%02X, value=0x%08X (cmd=0x%02X)", rep.status,
221 rep.value, cmd);
222 return true;
223}

237template <typename CommType>
238bool TMC9660Bootloader<CommType>::sendCommandUART(uint8_t cmd, uint32_t value, uint32_t* reply) noexcept {
239 if (comm_.mode() != CommMode::UART) {
240 return false;
241 }
242 // At this point, we know comm_.mode() == CommMode::UART, so CommType must be a UART interface
243 // Call uartTransferBootloader directly on comm_ - if CommType is UART, it will have this method
244 // We use a helper function pointer to avoid template instantiation issues
245 BootloaderCommandUART tx{deviceAddr_, cmd, value};
246
247 TMC9660_LOG_DEBUG(comm_, 4, "TMC9660Bootloader",
248 "Sending UART bootloader command: cmd=0x%02X, value=0x%08X", cmd, value);
249
250 std::array<uint8_t, 8> txBuf, rxBuf;
251 tx.toBuffer(txBuf);
252
253 // Log transmitted bytes
254 // TMC9660_LOG_DEBUG(comm_,2, "TMC9660Bootloader", "[UART BL TX] %02X %02X %02X %02X %02X %02X
255 // %02X %02X",
256 // txBuf[0], txBuf[1], txBuf[2], txBuf[3], txBuf[4], txBuf[5], txBuf[6], txBuf[7]);
257
258 // UART: Send command and receive reply in same transaction (no delayed reply like SPI)
259 // Call the method directly on comm_ - since CommType inherits from UartCommInterface<CommType>,
260 // it has uartTransferBootloader() available through inheritance. The CRTP base class method
261 // will internally use static_cast<Derived*>(this) to call the derived implementation.
262 if (!comm_.uartTransferBootloader(txBuf, rxBuf)) {
263 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to send UART command (cmd=0x%02X)",
264 cmd);
265 return false;
266 }
267
268 // Log received bytes
269 // TMC9660_LOG_DEBUG(comm_,2, "TMC9660Bootloader", "[UART BL RX] %02X %02X %02X %02X %02X %02X
270 // %02X %02X",
271 // rxBuf[0], rxBuf[1], rxBuf[2], rxBuf[3], rxBuf[4], rxBuf[5], rxBuf[6], rxBuf[7]);
272
273 // Parse reply
274 BootloaderReplyUART rep = BootloaderReplyUART::fromBuffer(rxBuf);
275
276 // Verify CRC
277 if (!rep.verifyCRC(rxBuf)) {
278 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "UART reply CRC mismatch (cmd=0x%02X)", cmd);
279 return false;
280 }
281
282 if (!rep.isOK()) {
283 if (IsExpectedOptionalInfoFailure(cmd, value, rep.status)) {
284 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader",
285 "Optional bootloader info unavailable: status=0x%02X (cmd=0x%02X, query=0x%08X)",
286 rep.status, cmd, value);
287 } else {
288 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader",
289 "UART command failed: status=0x%02X (cmd=0x%02X)", rep.status, cmd);
290 }
291 return false;
292 }
293
294 if (reply)
295 *reply = rep.value;
296
297 TMC9660_LOG_DEBUG(comm_, 4, "TMC9660Bootloader",
298 "UART command successful: status=0x%02X, value=0x%08X (cmd=0x%02X)", rep.status,
299 rep.value, cmd);
300 return true;
301}

550template <typename CommType>
551bool TMC9660Bootloader<CommType>::otpBurnWithWorkaround(uint8_t page, uint8_t page_addr, OtpBurnResult* result,
552 uint32_t vdrv_wait_ms) noexcept {
553 if (!result) {
554 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "otpBurnWithWorkaround: null result pointer");
555 return false;
556 }
557
558 TMC9660_LOG_DEBUG(comm_, 2, "TMC9660Bootloader",
559 "Starting OTP burn with Erratum 1 workaround (page=%d, addr=0x%02X)", page,
560 page_addr);
561
562 // Step 1: Send SET_BANK, value 0
563 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", "Step 1: Setting bank to 0");
564 if (!setBank(0)) {
565 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to set bank 0");
566 *result = OtpBurnResult::createError(OtpBurnError::BURN_PROCEDURE_FAILED);
567 return false;
568 }
569
570 // Step 2: Send SET_ADDRESS, value 0x4801B010
571 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", "Step 2: Setting address to 0x4801B010");
572 if (!setAddress(0x4801B010)) {
573 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to set address 0x4801B010");
574 *result = OtpBurnResult::createError(OtpBurnError::BURN_PROCEDURE_FAILED);
575 return false;
576 }
577
578 // Step 3: Send READ_32
579 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", "Step 3: Reading 32-bit value");
580 uint32_t read_value;
581 if (!read32(&read_value)) {
582 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to read 32-bit value");
583 *result = OtpBurnResult::createError(OtpBurnError::BURN_PROCEDURE_FAILED);
584 return false;
585 }
586
587 // Step 4: Clear bit 0 of the read value (0x00000001)
588 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", "Step 4: Clearing bit 0 (read=0x%08X)",
589 read_value);
590 uint32_t modified_value = read_value & ~0x00000001;
591
592 // Step 5: Send WRITE_32 with the modified read value
593 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", "Step 5: Writing modified value 0x%08X",
594 modified_value);
595 if (!write32(modified_value)) {
596 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to write modified value");
597 *result = OtpBurnResult::createError(OtpBurnError::BURN_PROCEDURE_FAILED);
598 return false;
599 }
600
601 // Step 6: Wait for VDRV voltage to drop below 8.4V
602 TMC9660_LOG_DEBUG(comm_, 2, "TMC9660Bootloader",
603 "Step 6: Waiting %dms for VDRV voltage to drop below 8.4V", vdrv_wait_ms);
604 comm_.delayMs(vdrv_wait_ms);
605
606 // Step 7: Send OTP_BURN
607 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", "Step 7: Sending OTP_BURN command");
608 uint32_t value = (static_cast<uint32_t>(page_addr) << 8) | page;
609 uint32_t reply = 0;
610
611 if (!sendCommand(static_cast<uint8_t>(BootloaderCommand::OTP_BURN), value, &reply)) {
612 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to send OTP_BURN command");
613 *result = OtpBurnResult::createError(OtpBurnError::BURN_PROCEDURE_FAILED);
614 return false;
615 }
616
617 // Parse the result
618 if (reply == 0) {
619 TMC9660_LOG_DEBUG(comm_, 2, "TMC9660Bootloader", "OTP burn command completed successfully");
620 *result = OtpBurnResult::createSuccess();
621 } else {
622 int8_t error_code = static_cast<int8_t>(reply);
623 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "OTP burn command failed with error code: %d",
624 error_code);
625 *result = OtpBurnResult::createError(static_cast<OtpBurnError>(error_code));
626 }
627
628 return true;
629}

631template <typename CommType>
632
633bool TMC9660Bootloader<CommType>::checkOtpBurnStatus(bool* result) noexcept {
634 if (!result) {
635 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "checkOtpBurnStatus: null result pointer");
636 return false;
637 }
638
639 TMC9660_LOG_DEBUG(comm_, 2, "TMC9660Bootloader",
640 "Checking OTP burn status using Erratum 1 workaround");
641
642 // Step 1: Configure clock settings to have PLL active, SYS_CLK_DIV set to 3 (15MHz)
643 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader",
644 "Step 1: Configuring clock for 15MHz system clock");
645 if (!setBank(5)) { // CONFIG bank
646 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to set CONFIG bank");
647 return false;
648 }
649
650 // Read current clock configuration
651 if (!setAddress(0x00020018)) { // Clock config offset
652 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to set clock config address");
653 return false;
654 }
655
656 uint32_t clock_config;
657 if (!read32(&clock_config)) {
658 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to read clock configuration");
659 return false;
660 }
661
662 // Set SYS_CLK_DIV to 3 (15MHz system clock)
663 uint32_t modified_clock_config = (clock_config & ~0x03000000) | (3 << 24);
664 if (!write32(modified_clock_config)) {
665 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader",
666 "Failed to write modified clock configuration");
667 return false;
668 }
669
670 // Wait for clock configuration to take effect
671 comm_.delayMs(100);
672
673 // Step 2: Send SET_BANK, value 0
674 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", "Step 2: Setting bank to 0");
675 if (!setBank(0)) {
676 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to set bank 0");
677 return false;
678 }
679
680 // Step 3: Send SET_ADDRESS, value 0x48020014
681 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", "Step 3: Setting address to 0x48020014");
682 if (!setAddress(0x48020014)) {
683 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to set address 0x48020014");
684 return false;
685 }
686
687 // Step 4: Send READ_16
688 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", "Step 4: Reading 16-bit value");
689 uint16_t read_value;
690 if (!read16(&read_value)) {
691 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to read 16-bit value");
692 return false;
693 }
694
695 // Step 5: Check if read value is 0x80 or 0x84 (successful burn)
696 *result = (read_value == 0x80) || (read_value == 0x84);
697 TMC9660_LOG_DEBUG(comm_, 2, "TMC9660Bootloader", "OTP burn status check: read=0x%04X, success=%s",
698 read_value, *result ? "true" : "false");
699
700 // Step 6: Set SYS_CLK_DIV back to 0 (40MHz)
701 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", "Step 6: Restoring clock to 40MHz");
702 if (!setBank(5)) { // CONFIG bank
703 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to set CONFIG bank for clock restore");
704 return false;
705 }
706
707 if (!setAddress(0x00020018)) { // Clock config offset
708 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader",
709 "Failed to set clock config address for restore");
710 return false;
711 }
712
713 if (!write32(clock_config)) { // Restore original clock config
714 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader",
715 "Failed to restore original clock configuration");
716 return false;
717 }
718
719 // Wait for clock configuration to take effect
720 comm_.delayMs(100);
721
722 return true;
723}

1008template <typename CommType>
1009bool TMC9660Bootloader<CommType>::applyConfiguration(const BootloaderConfig& cfg,
1010 bool fail_on_verify_error) noexcept {
1011 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", "Starting bootloader configuration application");
1012
1013 if (!setBank(5)) {
1014 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to set bank 5");
1015 return false;
1016 }
1017 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", "Successfully set bank 5 (CONFIG memory bank)");
1018
1019 // LDO configuration (offset 0x00)
1020 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", "Configuring LDO settings");
1021 if (!setAddress(bootaddr::LDO_CONFIG)) {
1022 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to set LDO config register");
1023 return false;
1024 }
1025 uint16_t ldo = 0;
1026 ldo |= static_cast<uint16_t>(cfg.ldo.vext1) & 0x3; // Bits 0-1: VEXT1
1027 ldo |= (static_cast<uint16_t>(cfg.ldo.vext2) & 0x3) << 2; // Bits 2-3: VEXT2
1028 ldo |= (static_cast<uint16_t>(cfg.ldo.slope_vext1) & 0x3) << 4; // Bits 4-5: SS_VEXT1
1029 ldo |= (static_cast<uint16_t>(cfg.ldo.slope_vext2) & 0x3) << 6; // Bits 6-7: SS_VEXT2
1030 ldo |= (cfg.ldo.ldo_short_fault ? 1u : 0u) << 8; // Bit 8: LDO_SHORT_FAULT
1031 if (!write16(ldo)) {
1032 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to write LDO config (0x%04X)", ldo);
1033 return false;
1034 }
1035 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", "LDO configured (VEXT1: %d, VEXT2: %d)",
1036 static_cast<int>(cfg.ldo.vext1), static_cast<int>(cfg.ldo.vext2));
1037
1038 // Verify LDO configuration was written correctly
1039 if (!readAndVerify16(ldo, "LDO config")) {
1040 if (fail_on_verify_error) {
1041 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "LDO configuration verification failed");
1042 return false;
1043 } else {
1044 TMC9660_LOG_DEBUG(comm_, 1, "TMC9660Bootloader",
1045 "⚠️ LDO configuration verification failed (continuing)");
1046 }
1047 }
1048
1049 // ⚠️ CRITICAL: Clock configuration (offset 0x18) - MUST BE DONE RIGHT BEFORE MOTOR CONTROL START!
1050 // Clock reconfiguration takes time and the TMC9660 cannot respond during this process.
1051 // This must happen just before starting motor control to ensure proper timing.
1052 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader",
1053 "Configuring clock settings (FINAL STEP BEFORE MOTOR CONTROL)");
1054 if (!setAddress(bootaddr::CLOCK_CONFIG)) {
1055 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to set clock config register");
1056 return false;
1057 }
1058
1059 // Test: Read current clock configuration before writing
1060 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", "Reading current clock configuration...");
1061 uint32_t current_clock;
1062 if (!read32(&current_clock)) {
1063 TMC9660_LOG_DEBUG(comm_, 1, "TMC9660Bootloader",
1064 "⚠️ Failed to read current clock config (continuing)");
1065 } else {
1066 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", "Current clock config: 0x%08X (PLL_STATUS=%s)",
1067 current_clock, (current_clock & (1u << 30)) ? "SET" : "CLEAR");
1068 }
1069 // ⚠️ CRITICAL: Bits 0-6 MUST be set to 99 (0x63) per datasheet requirement!
1070 // "RESERVED_1: Bits 0-6. Reserved. Must always stay set to 99."
1071 uint32_t clk = 99u; // Start with required reserved value
1072 clk |= (static_cast<uint32_t>(cfg.clock.use_external) & 0x1) << 8; // EXT_NOT_INT [0:6]
1073 clk |= (static_cast<uint32_t>(cfg.clock.xtal_drive) & 0x7) << 9; // XTAL_CFG [7:9]
1074 clk |= (cfg.clock.xtal_boost ? 1u : 0u) << 12; // XTAL_BOOST [12:12]
1075 clk |= (static_cast<uint32_t>(cfg.clock.ext_source_type) & 0x1) << 13; // EXT_NOT_XTAL [13:13]
1076 clk |= (static_cast<uint32_t>(cfg.clock.pll_selection) & 0x3) << 16; // PLL_OUT_SEL [16:17]
1077 clk |= (cfg.clock.rdiv & 0x1F) << 18; // RDIV [18:22]
1078 clk |= (static_cast<uint32_t>(cfg.clock.sysclk_div) & 0x3) << 23; // SYS_CLK_DIV [23:24]
1079 if (!write32(clk)) {
1080 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to write clock config (0x%08X)", clk);
1081 return false;
1082 }
1083 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader",
1084 "Clock configuration written (external: %s, xtal_drive: %d, rdiv: %d)",
1085 cfg.clock.use_external == bootcfg::ClockSource::External ? "yes" : "no",
1086 static_cast<int>(cfg.clock.xtal_drive), cfg.clock.rdiv);
1087
1088 // ⚠️ CRITICAL: Wait for clock reconfiguration to complete
1089 // According to datasheet: "Any change to the clock configuration takes multiple milliseconds to
1090 // apply. The TMC9660 cannot respond to commands during clock reconfiguration."
1091 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader",
1092 "Waiting for clock reconfiguration to complete...");
1093
1094 // Fixed delay to allow clock reconfiguration to complete
1095 // The TMC9660 cannot respond to commands during this process
1096 comm_.delayMs(100); // 100ms should be sufficient for clock reconfiguration
1097
1098 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", "Clock reconfiguration delay completed");
1099
1100 // Verify clock config was written correctly and PLL_STATUS is SET
1101 uint32_t actual_clock;
1102 if (!read32(&actual_clock)) {
1103 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to read back clock config");
1104 if (fail_on_verify_error) {
1105 return false;
1106 } else {
1107 TMC9660_LOG_DEBUG(comm_, 1, "TMC9660Bootloader", "⚠️ Clock config read failed (continuing)");
1108 }
1109 } else {
1110 // First check: PLL_STATUS must be SET after clock reconfiguration
1111 if (!(actual_clock & (1u << 30))) {
1112 if (fail_on_verify_error) {
1113 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader",
1114 "❌ PLL_STATUS not set after clock reconfiguration: 0x%08X", actual_clock);
1115 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader",
1116 " Clock reconfiguration may have failed or needs more time");
1117 return false;
1118 } else {
1119 TMC9660_LOG_DEBUG(comm_, 1, "TMC9660Bootloader",
1120 "⚠️ PLL_STATUS not set after clock reconfiguration: 0x%08X (continuing)",
1121 actual_clock);
1122 }
1123 } else {
1124 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", "✅ PLL_STATUS confirmed SET: 0x%08X",
1125 actual_clock);
1126 }
1127
1128 // Second check: Verify the actual configuration matches expected (excluding PLL_STATUS)
1129 uint32_t expected_config = clk;
1130 uint32_t actual_config = actual_clock & ~(1u << 30); // Mask out PLL_STATUS bit for comparison
1131
1132 if (actual_config != expected_config) {
1133 if (fail_on_verify_error) {
1134 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader",
1135 "❌ Clock config verification failed: expected=0x%08X, actual=0x%08X",
1136 expected_config, actual_config);
1137 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader",
1138 " Full actual value: 0x%08X (PLL_STATUS=%s)", actual_clock,
1139 (actual_clock & (1u << 30)) ? "SET" : "CLEAR");
1140 return false;
1141 } else {
1142 TMC9660_LOG_DEBUG(
1143 comm_, 1, "TMC9660Bootloader",
1144 "⚠️ Clock config verification failed: expected=0x%08X, actual=0x%08X (continuing)",
1145 expected_config, actual_config);
1146 }
1147 } else {
1148 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader",
1149 "✅ Clock config verified: 0x%08X (PLL_STATUS=%s)", actual_clock,
1150 (actual_clock & (1u << 30)) ? "SET" : "CLEAR");
1151 }
1152 }
1153
1154 // UART addresses (offset 0x02)
1155 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", "Configuring UART addresses");
1156 if (!setAddress(bootaddr::UART_ADDR)) {
1157 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to set UART address register");
1158 return false;
1159 }
1160 uint16_t addr_word = static_cast<uint16_t>(cfg.uart.device_address) |
1161 (static_cast<uint16_t>(cfg.uart.host_address) << 8);
1162 if (!write16(addr_word)) {
1163 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to write UART address word (0x%04X)",
1164 addr_word);
1165 return false;
1166 }
1167 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader",
1168 "UART addresses configured (device: %d, host: %d)", cfg.uart.device_address,
1169 cfg.uart.host_address);
1170
1171 // Verify UART addresses were written correctly
1172 if (!readAndVerify16(addr_word, "UART addresses")) {
1173 if (fail_on_verify_error) {
1174 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "UART addresses verification failed");
1175 return false;
1176 } else {
1177 TMC9660_LOG_DEBUG(comm_, 1, "TMC9660Bootloader",
1178 "⚠️ UART addresses verification failed (continuing)");
1179 }
1180 }
1181
1182 // RS485 delays
1183 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", "Configuring RS485 delays");
1184 if (!setAddress(bootaddr::RS485_DELAY)) {
1185 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to set RS485 delay register");
1186 return false;
1187 }
1188 uint16_t rs485 = static_cast<uint16_t>(cfg.rs485.txen_post_delay) |
1189 (static_cast<uint16_t>(cfg.rs485.txen_pre_delay) << 8);
1190 if (!write16(rs485)) {
1191 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to write RS485 delay config (0x%08X)",
1192 rs485);
1193 return false;
1194 }
1195 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", "RS485 delays configured (pre: %d, post: %d)",
1196 cfg.rs485.txen_pre_delay, cfg.rs485.txen_post_delay);
1197
1198 // Verify RS485 delays were written correctly
1199 if (!readAndVerify16(rs485, "RS485 delays")) {
1200 if (fail_on_verify_error) {
1201 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "RS485 delays verification failed");
1202 return false;
1203 } else {
1204 TMC9660_LOG_DEBUG(comm_, 1, "TMC9660Bootloader",
1205 "⚠️ RS485 delays verification failed (continuing)");
1206 }
1207 }
1208
1209 // Communication configuration (UART/SPI/RS485) - SHARED REGISTER with SPI Flash
1210 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader",
1211 "Configuring communication settings (shared register with SPI flash)");
1212 if (!setAddress(bootaddr::COMM_CONFIG)) {
1213 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to set communication config register");
1214 return false;
1215 }
1216
1217 // Read current communication config before overwriting
1218 uint16_t current_comm = 0;
1219 if (read16(&current_comm)) {
1220 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", "Current COMM_CONFIG before overwrite: 0x%04X",
1221 current_comm);
1222 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", " Current Bit 0 (UART disabled): %d",
1223 (current_comm >> 0) & 0x1);
1224 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", " Current Bit 1 (SPI disabled): %d",
1225 (current_comm >> 1) & 0x1);
1226 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", " Current Bit 2 (SPI SELECT): %d",
1227 (current_comm >> 2) & 0x1);
1228 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", " Current Bit 3 (UART RX): %d",
1229 (current_comm >> 3) & 0x1);
1230 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", " Current Bit 4 (UART TX): %d",
1231 (current_comm >> 4) & 0x1);
1232 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", " Current Bits 5-6 (UART TXEN): %d",
1233 (current_comm >> 5) & 0x3);
1234 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", " Current Bits 7-9 (UART BAUD): %d",
1235 (current_comm >> 7) & 0x7);
1236 TMC9660_LOG_DEBUG(comm_, 3, "TMC9660Bootloader", " Current Bit 10 (SPI0 SCK): %d",
1237 (current_comm >> 10) & 0x1);
1238 } else {
1239 TMC9660_LOG_DEBUG(comm_, 1, "TMC9660Bootloader",
1240 "⚠️ Failed to read current COMM_CONFIG (continuing)");
1241 }
1242
1243 // Build the shared register value (offset 6) - 16-bit write (highest bit is 10)
1244 uint16_t comm = 0;
1245
1246 // UART Configuration
1247 comm |= (cfg.uart.disable_uart ? 1u : 0u) << 0; // BL_DISABLE_UART (bit 0)
1248 comm |= (static_cast<uint16_t>(cfg.uart.rx_pin) & 0x1)
1249 << 3; // BL_UART_RX (bit 3): 0=GPIO7, 1=GPIO1
1250 comm |= (static_cast<uint16_t>(cfg.uart.tx_pin) & 0x1)
1251 << 4; // BL_UART_TX (bit 4): 0=GPIO6, 1=GPIO0
1252 comm |= (static_cast<uint16_t>(cfg.rs485.txen_pin) & 0x3) << 5; // BL_UART_TXEN (bits 5-6)
1253 comm |= (static_cast<uint16_t>(cfg.uart.baud_rate) & 0x7) << 7; // BL_UART_BAUDRATE (bits 7-9)
1254
1255 // SPI Configuration
1256 comm |= (cfg.spiComm.disable_spi ? 1u : 0u) << 1; // BL_DISABLE_SPI (bit 1)
1257
1258 // ============================================================================
1259 // CRITICAL: Handle shared BL_SPI_SELECT (bit 2) and BL_SPI0_SCK (bit 10)
1260 // ============================================================================
1261 // These bits are SHARED between bootloader comm (offset 6) and flash (offset 10).
1262 // Per datasheet: "If both flash and bootloader SPI communication are used,
1263 // they must always be on separate SPI interfaces."
1264 //
1265 // Priority rules:
1266 // 1. If bootloader SPI enabled: bootloader config controls BL_SPI_SELECT
1267 // 2. If bootloader SPI disabled but flash enabled: flash config controls it
1268 // 3. Bootloader communication ALWAYS takes priority over flash
1269 // ============================================================================
1270
1271 // Determine which physical SPI interface each component wants to use
1272 bool bootloader_uses_spi0 = !cfg.spiComm.disable_spi &&
1273 (cfg.spiComm.boot_spi_iface == tmc9660::bootcfg::SPIInterface::SPI0);
1274 bool flash_uses_spi0 = cfg.spiFlash.enable_flash &&
1275 (cfg.spiFlash.flash_spi_iface == tmc9660::bootcfg::SPIInterface::SPI0);
1276
1277 // Validate configuration: both cannot use the same SPI interface
1278 if (!cfg.spiComm.disable_spi && cfg.spiFlash.enable_flash) {
1279 if (bootloader_uses_spi0 == flash_uses_spi0) {
1280 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "⚠️ CONFIGURATION ERROR:");
1281 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader",
1282 " Both bootloader and flash are configured for %s!",
1283 bootloader_uses_spi0 ? "SPI0" : "SPI1");
1284 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader",
1285 " Per datasheet: they MUST be on separate interfaces");
1286 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", " Current config:");
1287 TMC9660_LOG_DEBUG(
1288 comm_, 0, "TMC9660Bootloader", " - Bootloader SPI: %s (disable_spi=%d)",
1289 cfg.spiComm.boot_spi_iface == tmc9660::bootcfg::SPIInterface::SPI0 ? "SPI0" : "SPI1",
1290 cfg.spiComm.disable_spi ? 1 : 0);
1291 TMC9660_LOG_DEBUG(
1292 comm_, 0, "TMC9660Bootloader", " - Flash SPI: %s (enable_flash=%d)",
1293 cfg.spiFlash.flash_spi_iface == tmc9660::bootcfg::SPIInterface::SPI0 ? "SPI0" : "SPI1",
1294 cfg.spiFlash.enable_flash ? 1 : 0);
1295 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader",
1296 " Fix: Set one to SPI0 and the other to SPI1");
1297 return false;
1298 }
1299 }
1300
1301 TMC9660_LOG_DEBUG(comm_, 2, "TMC9660Bootloader", "SPI interface allocation:");
1302 TMC9660_LOG_DEBUG(comm_, 2, "TMC9660Bootloader", " - Bootloader: %s (enabled=%d)",
1303 bootloader_uses_spi0 ? "SPI0" : "SPI1", !cfg.spiComm.disable_spi ? 1 : 0);
1304 TMC9660_LOG_DEBUG(comm_, 2, "TMC9660Bootloader", " - Flash: %s (enabled=%d)",
1305 flash_uses_spi0 ? "SPI0" : "SPI1", cfg.spiFlash.enable_flash ? 1 : 0);
1306
1307 // ============================================================================
1308 // Set BL_SPI_SELECT (bit 2) with correct bit inversion logic
1309 // ============================================================================
1310 // CRITICAL: BL_SPI_SELECT has OPPOSITE meanings for bootloader vs flash!
1311 // For Bootloader: 0=SPI0, 1=SPI1
1312 // For Flash: 0=SPI1, 1=SPI0 (inverted!)
1313 //
1314 // Priority: Bootloader config takes precedence over flash config
1315 // ============================================================================
1316 uint8_t bl_spi_select_bit;

1417 // ============================================================================
1418 // NOTE: This register mirrors some bits from COMM_CONFIG (offset 6):

1420 // - Bit 2: SPI0_SCK_SELECT (mirrors BL_SPI0_SCK from COMM_CONFIG bit 10)
1421 //
1422 // These bits were determined above based on priority:
1423 // - If bootloader SPI enabled: bootloader config takes priority
1424 // - If bootloader SPI disabled but flash enabled: flash config is used
1425 //
1426 // This register (offset 10) contains:
1427 // - Bit 0: SPI_FLASH_EN
1428 // - Bit 1: SPI_IFACE_SELECT (mirrors COMM_CONFIG bit 2)
1429 // - Bit 2: SPI0_SCK_SELECT (mirrors COMM_CONFIG bit 10)
1430 // - Bits 3-7: SPI_FLASH_CS (chip select pin)
1431 // - Bits 8-11: SPI_FLASH_FREQ (frequency divider)
1432 // ============================================================================

1437 return false;

1451 << 1; // SPI_IFACE_SELECT (bit 1, mirrors COMM_CONFIG bit 2)
1452 flash |= (static_cast<uint16_t>(sck_pin_bit & 0x1))
1453 << 2; // SPI0_SCK_SELECT (bit 2, mirrors COMM_CONFIG bit 10)

2150template <typename CommType>
2151bool TMC9660Bootloader<CommType>::startMotorControl(bootcfg::BootMode boot_mode) noexcept {
2152 TMC9660_LOG_DEBUG(comm_, 2, "TMC9660Bootloader", "Starting motor control system (mode: %d)...",
2153 static_cast<int>(boot_mode));
2154
2155 // Set bank to CONFIG (bank 5)
2156 if (!setBank(5)) {
2157 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader",
2158 "Failed to set CONFIG bank before starting motor control");
2159 return false;
2160 }
2161
2162 // Set address to BOOT_CONFIG register (offset 0x08)
2163 if (!setAddress(bootaddr::BOOT_CONFIG)) {
2164 TMC9660_LOG_DEBUG(comm_, 0, "TMC9660Bootloader", "Failed to set BOOT_CONFIG address");
2165 return false;
2166 }
2167
2168 // Read current BOOT_CONFIG to preserve other settings
2169 // We'll use a dummy write to get the current value via SPI reply mechanism
2170 // Note: This is optional - if we don't care about preserving other bits, we can skip this
2171
2172 // Construct BOOT_CONFIG value with START_MOTOR_CTRL=1
2173 uint32_t boot_config = 0;
2174 boot_config |= static_cast<uint32_t>(boot_mode) & 0x3; // Bits 0-1: BOOT_MODE
2175 boot_config |= (1u << 12); // Bit 12: START_MOTOR_CTRL = 1
2176
2177 // Optional: Set BL_EXIT_FAULT to get FAULTN signal when exiting (helps debugging)
