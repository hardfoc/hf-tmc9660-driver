58static constexpr uint8_t crc8Bootloader(const uint8_t* data, size_t len) noexcept {
59 uint16_t crc = 0; // 9-bit register for polynomial division
60 const uint16_t POLY = 0x107; // x^8 + x^2 + x^1 + x^0 in 9-bit form
61
62 // Process each byte (bit-reversed, LSB-first)
63 for (size_t i = 0; i < len; i++) {
64 uint8_t byte_reversed = reverseByte(data[i]);
65
66 // Feed 8 bits into CRC register (MSB-first from reversed byte)
67 for (int bit = 7; bit >= 0; bit--) {
68 crc = (crc << 1) | ((byte_reversed >> bit) & 1);
69 if (crc & 0x100) { // If bit 8 is set
70 crc ^= POLY;
71 }
72 }
73 }
74
75 // Append 8 zero bits (flush remaining data through CRC)
76 for (int i = 0; i < 8; i++) {
77 crc <<= 1;
78 if (crc & 0x100) {
79 crc ^= POLY;
80 }
81 }
82
83 return static_cast<uint8_t>(crc & 0xFF);
84}
