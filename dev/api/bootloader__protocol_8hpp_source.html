339struct BootloaderCommandUART {
340 uint8_t deviceAddr;
341 uint8_t command;
342 uint32_t value;
343
345 void toBuffer(std::array<uint8_t, 8>& out) const noexcept {
346 out[0] = 0x55; // Sync byte
347 out[1] = deviceAddr;
348 out[2] = command;
349 out[3] = static_cast<uint8_t>(value >> 24); // MSB first
350 out[4] = static_cast<uint8_t>(value >> 16);
351 out[5] = static_cast<uint8_t>(value >> 8);
352 out[6] = static_cast<uint8_t>(value);
353 out[7] = crc8Bootloader(out.data(), 7); // CRC over first 7 bytes
354 }

401struct BootloaderReplyUART {
402 uint8_t hostAddr;
403 uint8_t deviceAddr;
404 uint8_t status;
405 uint32_t value;
406
408 static BootloaderReplyUART fromBuffer(const std::array<uint8_t, 8>& in) noexcept {
409 BootloaderReplyUART r;
410 // UART protocol: [HOST_ADDR] [DEVICE_ADDR] [STATUS] [VALUE(32)] [CRC8]
411 // (No SYNC byte in reply, unlike the request which has 0x55 as first byte)
412 r.hostAddr = in[0];
413 r.deviceAddr = in[1];
414 r.status = in[2];
415 r.value = (static_cast<uint32_t>(in[3]) << 24) | // MSB first
416 (static_cast<uint32_t>(in[4]) << 16) | (static_cast<uint32_t>(in[5]) << 8) |
417 static_cast<uint32_t>(in[6]);
418 // Note: in[7] is CRC8, should be verified
419 return r;
420 }
421
423 bool verifyCRC(const std::array<uint8_t, 8>& in) const noexcept {
424 uint8_t calculatedCRC = crc8Bootloader(in.data(), 7);
425 return calculatedCRC == in[7];
426 }
427
429 bool isOK() const noexcept {
430 return status == static_cast<uint8_t>(BootloaderStatus::OK);
431 }
432
434 BootloaderStatus getStatus() const noexcept {
435 return static_cast<BootloaderStatus>(status);
436 }
